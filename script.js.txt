let current='en';
function renderLangs(){
 const bar=document.getElementById('langBar');
 bar.innerHTML='';
 langs.forEach(l=>{
   const b=document.createElement('button');
   b.className='lang-btn'+(l.code===current?' active':'');
   b.textContent=l.native;
   b.onclick=()=>{current=l.code; updateTexts(); renderLangs();};
   bar.appendChild(b);
 });
}
function updateTexts(){
 const t=translations[current];
 document.getElementById('tagline').textContent=t.tagline;
 document.getElementById('sub').textContent=t.sub;
 document.getElementById('cta').textContent=t.cta;
 document.getElementById('aboutText').textContent=t.about;
}
function createParticles(){
 const bg=document.getElementById('bgCanvas');
 for(let i=0;i<18;i++){
   const p=document.createElement('div');
   p.className='particle';
   p.style.left=Math.random()*100+'%';
   p.style.animationDelay=(Math.random()*6)+'s';
   p.style.animationDuration=(4+Math.random()*4)+'s';
   bg.appendChild(p);
 }
}
function createEq(){
 const eq=document.getElementById('eq');
 for(let i=0;i<18;i++){
   const s=document.createElement('span');
   s.style.animationDelay=(Math.random()*0.6)+'s';
   s.style.animationDuration=(0.3+Math.random()*0.6)+'s';
   eq.appendChild(s);
 }
}
window.onload=()=>{
 renderLangs();
 updateTexts();
 createParticles();
 createEq();
};
